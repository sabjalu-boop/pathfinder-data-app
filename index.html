    <div class="max-w-md mx-auto p-4">

        <!-- 설정 패널 (GAS 웹 앱 URL 입력) -->

        <div id="settingsPanel" class="bg-white p-4 rounded-xl shadow-sm mb-4 hidden">

            <label class="block text-xs font-semibold text-gray-600 mb-1">Google Apps Script 웹 앱 URL</label>

            <input type="text" id="scriptUrl" placeholder="https://script.google.com/macros/s/.../exec" class="w-full border p-2 rounded-lg text-sm mb-2">

            <button onclick="saveUrl()" class="w-full bg-gray-800 text-white py-2 rounded-lg text-sm font-semibold">URL 저장</button>

        </div>

        <!-- 모임 정보 입력 카드 -->

        <div class="bg-white p-4 rounded-xl shadow-sm mb-4">

            <div class="mb-3">

                <label class="block text-xs font-semibold text-gray-600 mb-1">모드 선택</label>

                <select id="modeSelect" onchange="changeMode()" class="w-full border p-2 rounded-lg text-sm bg-white">

                    <option value="attendance">출석 체크</option>

                    <option value="activity">활동 카운트</option>

                </select>

            </div>

           

            <div id="activityNameContainer" class="mb-3 hidden">

                <label class="block text-xs font-semibold text-gray-600 mb-1">활동 항목명</label>

                <input type="text" id="activityName" placeholder="예: 봉사활동, 발표 등" class="w-full border p-2 rounded-lg text-sm">

            </div>

            <button onclick="loadMembers()" class="w-full bg-blue-100 text-blue-700 py-2 rounded-lg text-sm font-semibold">회원 명부 불러오기</button>

        </div>

        <!-- 회원 목록 영역 -->

        <div id="memberListContainer" class="space-y-3">

            <div class="text-center text-gray-400 py-10 text-sm">상단의 '회원 명부 불러오기'를 눌러주세요.</div>

        </div>

    </div>

    <!-- 하단 고정 저장 버튼 -->

    <div id="footerBar" class="fixed bottom-0 left-0 right-0 bg-white border-t p-3 max-w-md mx-auto hidden shadow-lg">

        <button onclick="submitData()" class="w-full bg-blue-600 text-white py-3 rounded-xl font-bold text-center">전송 및 저장하기</button>

    </div>

    <script>

        let members = [];

        window.onload = function() {

            const savedUrl = localStorage.getItem('GAS_URL');

            if (savedUrl) {

                document.getElementById('scriptUrl').value = savedUrl;

            } else {

                document.getElementById('settingsPanel').classList.remove('hidden');

            }

        };

        function toggleSettings() {

            const panel = document.getElementById('settingsPanel');

            panel.classList.toggle('hidden');

        }

        function saveUrl() {

            const url = document.getElementById('scriptUrl').value.trim();

            if (!url) {

                alert('URL을 입력해주세요.');

                return;

            }

            localStorage.setItem('GAS_URL', url);

            alert('저장되었습니다!');

            toggleSettings();

        }

        function changeMode() {

            const mode = document.getElementById('modeSelect').value;

            const actContainer = document.getElementById('activityNameContainer');

            if (mode === 'activity') {

                actContainer.classList.remove('hidden');

            } else {

                actContainer.classList.add('hidden');

            }

            if (members.length > 0) renderMembers();

        }

        async function loadMembers() {

            const url = localStorage.getItem('GAS_URL');

            if (!url) {

                alert('설정에서 구글 스크립트 웹 앱 URL을 먼저 입력해주세요.');

                document.getElementById('settingsPanel').classList.remove('hidden');

                return;

            }

            const container = document.getElementById('memberListContainer');

            container.innerHTML = '<div class="text-center text-gray-400 py-10 text-sm">불러오는 중...</div>';

            try {

                const response = await fetch(url);

                members = await response.json();

               

                if (members.length === 0) {

                    container.innerHTML = '<div class="text-center text-gray-400 py-10 text-sm">등록된 회원이 없습니다. 구글 시트를 확인해주세요.</div>';

                    return;

                }

                renderMembers();

                document.getElementById('footerBar').classList.remove('hidden');

            } catch (error) {

                alert('회원 명부를 불러오는데 실패했습니다: ' + error);

                container.innerHTML = '<div class="text-center text-red-400 py-10 text-sm">로드 실패. URL을 확인해주세요.</div>';

            }

        }

        function renderMembers() {

            const mode = document.getElementById('modeSelect').value;

            const container = document.getElementById('memberListContainer');

            container.innerHTML = '';

            members.forEach((m, index) => {

                const card = document.createElement('div');

                card.className = 'bg-white p-4 rounded-xl shadow-sm flex items-center justify-between';

                let controlHtml = '';

                if (mode === 'attendance') {

                    const isChecked = m.attendanceStatus === '출석' ? 'checked' : '';

                    controlHtml = `

                        <label class="relative inline-flex items-center cursor-pointer">

                            <input type="checkbox" id="att_${index}" ${isChecked} class="sr-only peer">

                            <div class="w-11 h-6 bg-gray-200 peer-focus:outline-none rounded-full peer peer-checked:after:translate-x-full peer-checked:after:border-white after:content-[''] after:absolute after:top-[2px] after:left-[2px] after:bg-white after:border-gray-300 after:border after:rounded-full after:h-5 after:w-5 after:transition-all peer-checked:bg-blue-600"></div>

                            <span class="ml-2 text-sm font-medium text-gray-700">출석</span>

                        </label>

                    `;

                } else {

                    const count = m.activityCount || 0;

                    controlHtml = `

                        <div class="flex items-center space-x-2">

                            <button onclick="adjustCount(${index}, -1)" class="w-8 h-8 bg-gray-100 rounded-lg font-bold text-gray-600">-</button>

                            <span id="cnt_{count}</span>

                            <button onclick="adjustCount(${index}, 1)" class="w-8 h-8 bg-blue-100 text-blue-600 rounded-lg font-bold">+</button>

                        </div>

                    `;

                }

                card.innerHTML = `

                    <div>

                        <div class="font-bold text-gray-800">${m.이름}</div>

                        <div class="text-xs text-gray-400">${m.회원ID}</div>

                    </div>

                    ${controlHtml}

                `;

                container.appendChild(card);

            });

        }

        function adjustCount(index, delta) {

            if (!members[index].activityCount) members[index].activityCount = 0;

            members[index].activityCount += delta;

            if (members[index].activityCount < 0) members[index].activityCount = 0;

            document.getElementById(`cnt_${index}`).innerText = members[index].activityCount;

        }

        async function submitData() {

            const url = localStorage.getItem('GAS_URL');

            const mode = document.getElementById('modeSelect').value;

            let payload = {};

            if (mode === 'attendance') {

                let records = [];

                members.forEach((m, index) => {

                    const isChecked = document.getElementById(`att_${index}`).checked;

                    records.push({

                        memberId: m.회원ID,

                        name: m.이름,

                        status: isChecked ? '출석' : '결석'

                    });

                });

                payload = { type: 'attendance', records: records };

            } else {

                const activityName = document.getElementById('activityName').value.trim();

                if (!activityName) {

                    alert('활동 항목명을 입력해주세요.');

                    return;

                }

                let records = [];

                members.forEach((m, index) => {

                    const count = m.activityCount || 0;

                    if (count > 0) {

                        records.push({

                            memberId: m.회원ID,

                            name: m.이름,

                            activityName: activityName,

                            count: count

                        });

                    }

                });

                if (records.length === 0) {

                    alert('카운트가 0보다 큰 회원이 없습니다.');

                    return;

                }

                payload = { type: 'activity', records: records };

            }

            try {

                const response = await fetch(url, {

                    method: 'POST',

                    body: JSON.stringify(payload)

                });

                const result = await response.json();

                if (result.status === 'success') {

                    alert('성공적으로 저장되었습니다!');

                } else {

                    alert('저장 실패: ' + result.message);

                }

            } catch (error) {

                alert('통신 중 오류가 발생했습니다: ' + error);

            }

        }

    </script>
